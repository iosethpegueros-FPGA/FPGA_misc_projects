#INSTRUCTIONS TO BUILD PROJECT IN WINDOWS
#set FPGA_ROOT environment variable to the location of your FIFO_uvm_tb_ver0 repo
#eg
set FPGA_ROOT=C:\projects\FIFO_uvm_tb_ver0
#actual windows cmd command 
set FPGA_ROOT=%cd%
#validate variable
echo %FPGA_ROOT%



#run vivado in bash mode
#set the path to your vivado instalation
#eg
set PATH=%PATH%;C:\Xilinx\Vivado\2024.1\bin 
#launch vivado
vivado -mode tcl


#creat project / build / generate bitfiles with the scripts under
%FPGA_ROOT%/scripts


#create project
vivado -mode batch -source %FPGA_ROOT%\scripts\create_FIFO_uvm_tb_project.tcl

# vivado will throw an UVM fatal error due to not having a specific test to run, this is the windows BKM  for every UVM  environment 
vivado -mode batch -source %FPGA_ROOT%\scripts\simulate_FIFO_uvm_tb.tcl

# run any valid testName, tcl will show the right test list
vivado -mode batch -source %FPGA_ROOT%\scripts\run_tests_FIFO_uvm_tb.tcl -tclargs testName
